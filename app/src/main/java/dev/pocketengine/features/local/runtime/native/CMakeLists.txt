cmake_minimum_required(VERSION 3.22.1)
project(pocketworld_process LANGUAGES CXX)
add_library(pocketprocess SHARED process_supervisor.cpp pty_supervisor.cpp workspace_file_access.cpp)
target_compile_features(pocketprocess PRIVATE cxx_std_17)
target_compile_options(pocketprocess PRIVATE -Wall -Wextra -Werror)
target_link_options(pocketprocess PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384")
set_target_properties(pocketprocess PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
