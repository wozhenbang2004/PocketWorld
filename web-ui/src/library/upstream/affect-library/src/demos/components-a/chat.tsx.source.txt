export function WritingAssistant(p:DemoProps){const reduce=useReducedMotion();const [text,setText]=useState(''),[messages,setMessages]=useState<{role:string;text:string}[]>([]),[busy,setBusy]=useState(false),[voice,setVoice]=useState(false),[reply,setReply]=useState('');const timer=useTimers();const generation=useRef(0);const ask=(s:string)=>{if(!s.trim()||busy)return;const version=++generation.current;setMessages(m=>[...m,{role:'user',text:s}]);setText('');setBusy(true);setReply('');const full=s.toLowerCase().includes('short')?'Here’s a shorter version: “Thanks for the update.
