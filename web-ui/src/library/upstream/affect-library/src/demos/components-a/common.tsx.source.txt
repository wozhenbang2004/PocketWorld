/** Animate retained logical cards between their previous and next layout positions. */
export function useFlip(elementRef:{current:HTMLElement|null}) {
 const previous=useRef(new Map<string,{left:number;top:number}>());
 const animations=useRef<Animation[]>([]);
 useLayoutEffect(()=>{
  animations.current.forEach(a=>a.cancel());animations.current=[];
  const elements=elementRef.current?.querySelectorAll<HTMLElement>('[data-flip-id]');
  if(!elements)return;
  const next=new Map<string,{left:number;top:number}>();
  const reduce=window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  elements.forEach(el=>{const id=el.dataset.flipId!;const rect=el.getBoundingClientRect();next.set(id,{left:rect.left,top:rect.top});const old=previous.current.get(id);if(old&&!reduce){const dx=old.left-rect.left,dy=old.top-rect.top;if(Math.abs(dx)+Math.abs(dy)>1)animations.current.push(el.animate([{transform:`translate(${dx}px,${dy}px)`},{transform:'none'}],{duration:440,easing:'cubic-bezier(.22,1,.36,1)'}));}});
  previous.current=next;
 });
 useEffect(()=>()=>animations.current.forEach(a=>a.cancel()),[]);
}
