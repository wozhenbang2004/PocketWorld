/** A dependency-free, reusable progress display. All labels enter via textContent. */
export class AgentStepper extends HTMLElement {
  static observedAttributes = ['language'];

  constructor() {
    super();
    this.attachShadow({mode:'open'});
    this._steps = this._validate(DEMO_STEPS);
    this._elapsed = 0;
    this._playing = false;
    this._rate = 1;
    this._raf = 0;
    this._last = 0;
    this._lastState = '';
  }

  connectedCallback() {
    this._build();
    this._observer = new ResizeObserver(() => this._drawSpine());
    this._observer.observe(this._panel);
    this._rows.forEach(row => this._observer.observe(row.group));
    if (this.hasAttribute('autoplay')) this.play();
  }

  disconnectedCallback() {
    this.pause();
    this._observer?.disconnect();
  }

  attributeChangedCallback() {
    if (this.isConnected && this._rows) this._render(true);
  }

  get steps() { return structuredClone(this._steps); }
  set steps(value) {
    const steps = this._validate(value);
    this.pause();
    this._steps = steps;
    this._elapsed = 0;
    this._lastState = '';
    if (this.isConnected) {
      this._observer?.disconnect();
      this._build();
      this._observer?.observe(this._panel);
      this._rows.forEach(row => this._observer?.observe(row.group));
    }
  }

  get duration() { return this._steps.reduce((sum, step) => sum + (step.duration || 0), 0); }
  get elapsed() { return this._elapsed; }
  get playing() { return this._playing; }
  get playbackRate() { return this._rate; }
  set playbackRate(value) {
    if (!Number.isFinite(value) || value <= 0 || value > 16) throw new RangeError('playbackRate must be > 0 and <= 16');
    this._rate = value;
  }

  play() {
    if (this._playing || this._elapsed >= this.duration || !this.isConnected) return;
    this._playing = true;
    this.removeAttribute('paused');
    this._last = 0;
    this._render();
    const tick = now => {
      if (!this._playing) return;
      if (this._last) this._elapsed = Math.min(this.duration, this._elapsed + Math.min((now-this._last)/1000, .1)*this._rate);
      this._last = now;
      this._render();
      if (this._elapsed < this.duration) this._raf = requestAnimationFrame(tick);
      else { this._playing = false; this.setAttribute('paused',''); this._render(); }
    };
    this._raf = requestAnimationFrame(tick);
  }

  pause() {
    this._playing = false;
    this.setAttribute('paused','');
    cancelAnimationFrame(this._raf);
    this._last = 0;
    if (this._rows) this._render();
  }

  reset({autoplay=false} = {}) {
    this.pause();
    this._elapsed = 0;
    this._render(true);
    if (autoplay) this.play();
  }

  seek(seconds) {
    if (!Number.isFinite(seconds)) throw new TypeError('seek expects finite seconds');
    this._elapsed = Math.min(this.duration, Math.max(0, seconds));
    if (this._elapsed === this.duration) this.pause();
    this._last = 0;
    this._render(true);
  }

  _validate(value) {
    if (!Array.isArray(value) || value.length === 0) throw new TypeError('steps must be a non-empty array');
    const result = structuredClone(value);
    const ids = new Set();
    result.forEach((step, i) => {
      if (!step.id || ids.has(step.id) || typeof step.label !== 'string') throw new TypeError('Every step needs a unique id and label');
      ids.add(step.id);
      if (step.requiresApproval) {
        if (i !== result.length-1) throw new TypeError('An approval gate must be the final step');
        step.duration = 0;
        step.tasks = [];
      } else {
        if (!Number.isFinite(step.duration) || step.duration <= 0) throw new RangeError('Step duration must be positive');
        step.tasks = step.tasks || [];
        step.tasks.forEach((task, j) => {
          if (typeof task.label !== 'string' || !Number.isFinite(task.at) || task.at < 0 || task.at >= step.duration || (j && task.at <= step.tasks[j-1].at)) throw new TypeError('Tasks need labels and increasing at times within their step');
        });
      }
    });
    return result;
  }

  _build() {
    this.shadowRoot.replaceChildren();
    const style = document.createElement('style');
    style.textContent = AGENT_STEPPER_CSS;
    this._panel = document.createElement('div');
    this._panel.className = 'panel';
    this._panel.setAttribute('role','list');
    this._spine = document.createElementNS(SVG_NS,'svg');
    this._spine.classList.add('spine');
    this._spine.setAttribute('aria-hidden','true');
    this._panel.append(this._spine);
    this._rows = this._steps.map(step => {
      const group = document.createElement('div'); group.className = 'step'; group.setAttribute('role','listitem');
      const row = document.createElement('div'); row.className = 'row';
      const node = document.createElement('span'); node.className = 'node'; node.setAttribute('aria-hidden','true');
      const label = document.createElement('span'); label.className = 'label';
      const time = document.createElement('span'); time.className = 'time'; time.setAttribute('aria-hidden','true');
      const digitWindow = document.createElement('span'); digitWindow.className = 'digit-window';
      const unit = document.createElement('span'); unit.textContent = 's';
      time.append(digitWindow,unit);
      row.append(node,label,time);
      if (step.requiresApproval) { const badge = document.createElement('span'); badge.className='badge'; row.append(badge); }
      const children = document.createElement('div'); children.className='subtasks'; children.setAttribute('aria-hidden','true');
      const clip = document.createElement('div'); clip.className='clip';
      const top = document.createElement('div'); top.className='spacer'; clip.append(top);
      const tasks = step.tasks.map(() => {
        const item = document.createElement('div'); item.className='subtask';
        const inner = document.createElement('div'); inner.className='clip';
        const subrow = document.createElement('div'); subrow.className='subrow';
        const subnode = document.createElement('span'); subnode.className='subnode';
        const sublabel = document.createElement('span'); sublabel.className='sublabel';
        subrow.append(subnode,sublabel); inner.append(subrow); item.append(inner); clip.append(item);
        return {item,subnode,sublabel};
      });
      const bottom = document.createElement('div'); bottom.className='spacer'; clip.append(bottom); children.append(clip);
      group.append(row,children); this._panel.append(group);
      return {group,row,node,label,time,digitWindow,children,tasks,status:'',second:-1};
    });
    this._live = document.createElement('div'); this._live.className='sr-only'; this._live.setAttribute('role','status'); this._live.setAttribute('aria-live','polite');
    this.shadowRoot.append(style,this._panel,this._live);
    this.setAttribute('paused','');
    this._render(true);
    requestAnimationFrame(() => this._drawSpine());
  }

  _text(item, key='label') {
    return this.getAttribute('language') === 'zh' ? (item[key+'Zh'] || item[key] || item.label) : (item[key] || item.label);
  }

  _render(force=false) {
    if (!this._rows) return;
    const zh = this.getAttribute('language') === 'zh';
    let geometryChanged = force;
    this._panel.setAttribute('aria-label',zh?'任务进度':'Task progress');
    let start = 0, current = -1;
    this._steps.forEach((step,i) => {
      const r = this._rows[i];
      const end = start + step.duration;
      const status = step.requiresApproval ? 'locked' : this._elapsed >= end ? 'complete' : this._elapsed >= start ? 'running' : 'pending';
      if (status === 'running') current = i;
      if (r.status !== status) { r.node.innerHTML = glyphs[status]; r.status = status; r.group.dataset.status=status; geometryChanged=true; }
      if (r.label.textContent !== this._text(step)) r.label.textContent=this._text(step);
      const second = status === 'complete' ? step.duration : Math.floor(Math.max(0,this._elapsed-start));
      if (!step.requiresApproval && (r.second !== second || force)) {
        const old = r.digitWindow.querySelector('.digit:not(.out)');
        r.digitWindow.querySelectorAll('.out').forEach(el=>el.remove());
        if (old) { old.classList.add('out'); old.addEventListener('animationend',()=>old.remove(),{once:true}); setTimeout(()=>old.remove(),300); }
        const digit=document.createElement('span'); digit.className='digit'; digit.textContent=String(second); r.digitWindow.append(digit); r.second=second;
      }
      const statusLabel=zh?{pending:'待执行',running:this._playing?'执行中':'已暂停',complete:'已完成',locked:'需先确认'}[status]:{pending:'Pending',running:this._playing?'Running':'Paused',complete:'Complete',locked:'Asks first'}[status];
      r.group.setAttribute('aria-label',`${this._text(step)}: ${statusLabel}${status==='running'||status==='complete'?`, ${second}${zh?'秒':' seconds'}`:''}`);
      if (step.requiresApproval) r.row.querySelector('.badge').textContent=zh?'先确认':'Asks first';
      const local = Math.max(0,this._elapsed-start);
      const activeTask = step.tasks.findLastIndex(task=>local>=task.at);
      r.children.classList.toggle('open',status==='running' && activeTask>=0);
      r.children.setAttribute('aria-hidden',status==='running'?'false':'true');
      r.tasks.forEach((t,j)=>{
        const shown = j<=activeTask;
        t.item.classList.toggle('shown',shown);
        t.item.classList.toggle('current',j===activeTask);
        const text=this._text(step.tasks[j],j<activeTask?'doneLabel':'label');
        if(t.sublabel.textContent !== text) t.sublabel.textContent=text;
        t.item.setAttribute('aria-hidden',!shown?'true':'false');
      });
      start=end;
    });
    const state = this._elapsed >= this.duration ? (this._steps.at(-1).requiresApproval?'awaiting-approval':'complete') : this._playing?'playing':'paused';
    this.dataset.state=state;
    if(geometryChanged) requestAnimationFrame(()=>this._drawSpine());
    const stateKey=`${state}:${current}`;
    if (stateKey !== this._lastState || force) {
      this._lastState=stateKey;
      this._live.textContent=current>=0?`${this._text(this._steps[current])} — ${zh?(state==='playing'?'执行中':'已暂停'):(state==='playing'?'Running':'Paused')}`:state==='awaiting-approval'?(zh?'已完成准备，等待确认。未发送任何内容。':'Preparation complete. Waiting for approval; nothing has been sent.'):(zh?'所有任务已完成。':'All tasks complete.');
      this.dispatchEvent(new CustomEvent('runstatechange',{bubbles:true,composed:true,detail:{state,currentIndex:current,elapsed:this._elapsed,duration:this.duration}}));
    }
  }

  _drawSpine() {
    if (!this.isConnected || !this._rows?.length) return;
    const panel=this._panel.getBoundingClientRect();
    const centers=this._rows.map(r=>{const b=r.node.getBoundingClientRect();return{x:b.left-panel.left+b.width/2,y:b.top-panel.top+b.height/2};});
    this._spine.setAttribute('viewBox',`0 0 ${panel.width} ${panel.height}`);
    this._spine.replaceChildren();
    for(let i=0;i<centers.length-1;i++) {
      const a=centers[i],b=centers[i+1],r=this._rows[i];
      const childHeight=r.children.getBoundingClientRect().height;
      const offset=Math.min(13,Math.max(0,childHeight/2));
      const x=a.x, inset=x+offset;
      const top=a.y+15, bottom=b.y-15;
      const bend=Math.min(9,Math.max(0,(bottom-top)/3));
      const d=offset>.1 && bottom>top ? `M ${x} ${a.y} V ${top-bend} C ${x} ${top+2} ${inset} ${top-2} ${inset} ${top+bend} V ${bottom-bend} C ${inset} ${bottom+2} ${x} ${bottom-2} ${x} ${bottom+bend} V ${b.y}` : `M ${x} ${a.y} V ${b.y}`;
      const path=document.createElementNS(SVG_NS,'path'); path.setAttribute('d',d);
      if(r.status==='complete') path.classList.add('complete');
      this._spine.append(path);
    }
  }
}

if (!customElements.get('agent-stepper')) customElements.define('agent-stepper',AgentStepper);
