<script>
(()=>{
  const root=document.getElementById('josh-neutral-review');
  const button=root.querySelector('button');
  const result=root.querySelector('output');
  const reduced=window.matchMedia('(prefers-reduced-motion: reduce)');
  let press=0,strain=0,frame=0,owner=null,inside=false,count=0,suppressClick=false;
  let down=false;
  const paint=(value,compression)=>{
    press=value;strain=compression;
    button.style.setProperty('--press',String(value));
    button.style.setProperty('--strain',String(compression));
  };
  const stop=()=>{if(frame)cancelAnimationFrame(frame);frame=0;};
  const easeOut=t=>1-Math.pow(1-t,3);
  const move=pressed=>{
    if(down===pressed)return;
    down=pressed;button.dataset.state=pressed?'pressed':'released';stop();
    if(reduced.matches){paint(pressed?1:0,pressed ? .58 : 0);return;}
    const from=press,fromStrain=strain,start=performance.now();
    const duration=pressed?78:210;
    const tick=now=>{
      const t=Math.min(1,(now-start)/duration);
      let value,compression;
      if(pressed){
        value=t<.68?from+(1.03-from)*easeOut(t/.68):1.03-.03*easeOut((t-.68)/.32);
        compression=t<.68?fromStrain+(1-fromStrain)*easeOut(t/.68):1-.42*easeOut((t-.68)/.32);
      }else{
        value=t<.72?from+(-.065-from)*easeOut(t/.72):-.065+.065*easeOut((t-.72)/.28);
        compression=t<.72?fromStrain+(-.08-fromStrain)*easeOut(t/.72):-.08+.08*easeOut((t-.72)/.28);
      }
      paint(value,compression);
      if(t<1)frame=requestAnimationFrame(tick);
      else{frame=0;paint(pressed?1:0,pressed ? .58 : 0);button.dataset.state=pressed?'pressed':'idle';}
    };
    frame=requestAnimationFrame(tick);
  };
  const contains=event=>{
    const rect=button.getBoundingClientRect();
    return event.clientX>=rect.left&&event.clientX<=rect.right&&event.clientY>=rect.top&&event.clientY<=rect.bottom;
  };
  const cancel=()=>{
    if(owner!==null)suppressClick=true;
    const previous=owner;owner=null;inside=false;move(false);
    if(typeof previous==='number'&&button.hasPointerCapture(previous))button.releasePointerCapture(previous);
  };
  button.addEventListener('pointerdown',event=>{
    if(!event.isPrimary||event.button!==0||owner!==null)return;
    suppressClick=false;owner=event.pointerId;inside=true;
    button.setPointerCapture(event.pointerId);move(true);
  });
  button.addEventListener('pointermove',event=>{
    if(owner!==event.pointerId)return;
    inside=contains(event);move(inside);
  });
  button.addEventListener('pointerup',event=>{
    if(owner!==event.pointerId)return;
    inside=contains(event);suppressClick=!inside;owner=null;move(false);
    if(button.hasPointerCapture(event.pointerId))button.releasePointerCapture(event.pointerId);
  });
  button.addEventListener('pointercancel',event=>{if(owner===event.pointerId)cancel();});
  button.addEventListener('lostpointercapture',event=>{if(owner===event.pointerId)cancel();});
  button.addEventListener('keydown',event=>{
    if(event.key==='Escape'){event.preventDefault();cancel();return;}
    if(event.key!==' '&&event.key!=='Enter')return;
    event.preventDefault();
    if(event.repeat||owner!==null)return;
    suppressClick=false;owner=event.key;move(true);
  });
  button.addEventListener('keyup',event=>{
    if(event.key!==' '&&event.key!=='Enter')return;
    event.preventDefault();
    if(owner!==event.key)return;
    owner=null;move(false);button.click();
  });
  button.addEventListener('click',event=>{
    if(suppressClick){event.preventDefault();suppressClick=false;return;}
    count+=1;result.textContent='已按下 '+count+' 次';
  });
  button.addEventListener('blur',cancel);
  window.addEventListener('blur',cancel);
  document.addEventListener('visibilitychange',()=>{if(document.hidden)cancel();});
  reduced.addEventListener('change',()=>{stop();paint(down?1:0,down ? .58 : 0);button.dataset.state=down?'pressed':'idle';});
})();
</script>
