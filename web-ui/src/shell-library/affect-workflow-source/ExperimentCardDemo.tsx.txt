export function ExperimentCardDemo(p:DemoProps){
 const[expanded,setExpanded]=useState(false);const[pinned,setPinned]=useState(false);const[metric,setMetric]=useState('Conversion');const[winner,setWinner]=useState<number|null>(null);const[hover,setHover]=useState<number|null>(null);const rows=['Start free trial','Try it free','Get started – $0'];const values=metric==='Conversion'?[4.2,4.7,4.0]:metric==='Revenue'?[18.4,19.1,21.6]:[42.1,38.8,39.4];const lead=metric==='Revenue'?2:1;const lift=(i:number)=>i===0?'Base':(metric==='Bounce'?-1:1)*(values[i]-values[0])/values[0]*100;const open=expanded||pinned;
 return <Frame p={p}><Card className="cb-experiment2"><div onMouseEnter={()=>setExpanded(true)} onMouseLeave={()=>setExpanded(false)}><button className="cb-experiment-summary" aria-label={`Signup button copy, ${winner===null?'Running':'Completed'}, ${rows[winner??lead]} is ahead`} aria-expanded={open} onClick={()=>setPinned(v=>!v)}><span className="cb-experiment-mark"><i/><i/><i/></span><span className="cb-grow"><strong>Signup button copy</strong><span className="cb-row"><Tag tone="green">● {winner===null?'Running':'Completed'}</Tag><small>Day 9 of 14</small></span></span><Tag tone="green">{String.fromCharCode(65+(winner??lead))} ↗ <RollingNumber value={Math.abs(Number(lift(winner??lead)))/100} className="cb-experiment-inner"><Segmented label="Metric" values={['Conversion','Revenue','Bounce']} value={metric} onChange={setMetric}/><div className="cb-experiment-variants" role="radiogroup" aria-label="Winner">{rows.map((row,i)=><button key={row} role="radio" aria-checked={winner===i} onClick={()=>setWinner(i)} onMouseEnter={()=>setHover(i)} onMouseLeave={()=>setHover(null)} data-highlight={hover===i||winner===i}><span className="cb-variant-letter" style={{background:['#eee','#e2efff','#ece6fa'][i],color:['#aaa','#7399c4','#9a83c3'][i]}}>{String.fromCharCode(65+i)}</span><span className="cb-grow"><span className="cb-row"><span className="cb-grow">{row}</span><strong><RollingNumber value={values[i]} tone={i===0?'muted':Number(lift(i))>0?'green':'red'}>{i===0?'Base':<>{Number(lift(i))>0?'+':'−'}<RollingNumber value={Math.abs(Number(lift(i)))/100} format={{style:'percent',minimumFractionDigits:1,maximumFractionDigits:1}}/></>}</Tag></span><span className="cb-variant-line" style={{background:['#bbb','#4b93ef','#8a6fd0'][i],width:`${Math.min(100,values[i]/Math.max(...values)*100)}%`}}/></span></button>)}</div><div className="cb-experiment-hint">{winner===null?'Pick a winner':<span className="cb-row"><Icon name="check" size={12}/>{rows[winner]} chosen<Button onClick={()=>setWinner(null)}>Undo</Button></span>}</div></div></div></div></Card></Frame>
}
