export function PromptLibraryDemo(p:DemoProps){
 const[search,setSearch]=useState('');const[active,setActive]=useState(1);const[allValues,setAllValues]=useState<Record<number,Record<string,string>>>(Object.fromEntries(templates.map(t=>[t.id,{...t.defaults}])));const[used,setUsed]=useState(false);const[copied,setCopied]=useState(false);const item=templates.find(x=>x.id===active)!;const values=allValues[active]||{};const filled=item.variables.filter(v=>values[v.key]?.trim()).length;const parts=item.body.split(/(\{.*?\})/g);const assembled=item.body.replace(/\{(.*?)\}/g,(_,key)=>values[key]||`{${key}}`);const filtered=templates.filter(t=>(t.name+' '+t.category).toLowerCase().includes(search.toLowerCase()));const categoryTone=({Notes:'blue',Product:'purple',Debug:'orange',Sales:'green'} as Record<string,string>)[item.category];return <Frame p={p} wide><Card className="cb-prompt2"><div className="cb-prompt2-layout"><aside className="cb-prompt2-list"><div className="cb-row cb-prompt2-title"><strong>Prompts</strong><small>4</small></div><div className="cb-prompt2-search"><Icon name="search" size={12}/><input type="search" placeholder="Search templates" aria-label="Search templates" value={search} onChange={e=>setSearch(e.target.value)}/></div><div role="listbox" aria-label="Templates">{filtered.map(t=><button key={t.id} role="option" aria-selected={active===t.id} onClick={()=>{setActive(t.id);setUsed(false);setCopied(false)}}><span className="cb-grow">{t.name}</span><Tag tone={({Notes:'blue',Product:'purple',Debug:'orange',Sales:'green'} as Record<string,string>)[t.category]}>{t.category}</Tag></button>)}{!filtered.length&&<div className="cb-empty">No templates found</div>}</div></aside><div className="cb-prompt2-detail"><div className="cb-row cb-prompt2-heading"><strong className="cb-grow">{item.name}</strong><Tag tone={categoryTone}>{item.category}</Tag><span className="cb-variable-progress">{item.variables.map(v=><i key={v.key} data-filled={!!values[v.key]?.trim()}/>)}</span><small>{filled} of {item.variables.length} filled</small></div><p className="cb-prompt-sentence">{parts.map((part,i)=>{if(!part.startsWith('{'))return <span key={i}>{part}</span>;const key=part.slice(1,-1);const variable=item.variables.find(v=>v.key===key)!;return <input key={active+'-'+key} className="cb-inline-variable" aria-label={key+' – '+variable.hint} title={variable.hint} placeholder={key} value={values[key]||''} style={{width:Math.min(230,Math.max(key.length,(values[key]||'').length)*6.2+14)+'px'}} onChange={e=>{setAllValues(v=>({...v,[active]:{...v[active],[key]:e.target.value}}));setUsed(false)}}/>})}</p><small className="cb-prompt-preview-label">Preview</small><div className="cb-prompt2-preview" role="region" aria-label="Preview">{parts.map((part,i)=>part.startsWith('{')?<span key={i} className={values[part.slice(1,-1)]?'cb-preview-filled':'cb-preview-empty'}>{values[part.slice(1,-1)]||part}</span>:<span key={i}>{part}</span>)}</div>{used&&<small className="cb-prompt2-used" role="status">Prompt ready to use · prepared locally.</small>}</div></div><div className="cb-foot"><span>~{Math.max(1,Math.round(assembled.length/4.5))} tokens</span><div className="cb-row"><Button label="Copy prompt" onClick={()=>void navigator.clipboard?.writeText(assembled).then(()=>setCopied(true)).catch(()=>setCopied(false))}><Icon name={copied?'check':'copy'} size={11}/>{copied?'Copied':'Copy'}</Button><Button kind="primary" disabled={filled!==item.variables.length} onClick={()=>setUsed(true)}>Use prompt<Icon name="arrow" size={12}/></Button></div></div></Card></Frame>
}
